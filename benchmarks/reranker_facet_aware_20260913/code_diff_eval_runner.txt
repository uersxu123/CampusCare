+def aggregate_metrics(results: list[dict]) -> dict:
+    retrieval = [item for item in results if item.get("expectedDocumentKeys")]
+    usable_count = sum(len(item.get("usableDocumentKeys", [])) for item in results)
+    forbidden_count = sum(len(item.get("forbiddenUsable", [])) for item in results)
+    provenance = [item for item in results if item.get("usableDocumentKeys")]
+    negatives = [item for item in results if item.get("expectedGrade") == "NONE"]
+    citation_page_checks = sum(int(item.get("citationPageChecks", 0)) for item in results)
+    citation_page_matches = sum(int(item.get("citationPageMatches", 0)) for item in results)
+    return {
+        "recallAt5": _mean(item["recallAt5"] for item in retrieval),
+        "recallAt10": _mean(item["recallAt10"] for item in retrieval),
+        "mrr": _mean(item["reciprocalRank"] for item in retrieval),
+        "ndcgAt5": _mean(item["ndcgAt5"] for item in retrieval),
+        "forbiddenUsableEvidenceRate": forbidden_count / max(1, usable_count),
+        "gradeMacroF1": _not_applicable("FixedEvaluationStructuredClient 读取 golden label，仅用于编排合同"),
+        "gradeAccuracy": _not_applicable("FixedEvaluationStructuredClient 读取 golden label，仅用于编排合同"),
+        "negativeSufficientErrors": sum(item.get("actualGrade") == "SUFFICIENT" for item in negatives),
+        "provenanceCompleteness": _mean(1.0 if item.get("provenanceComplete") else 0.0 for item in provenance),
+        "citationPageAccuracy": (
+            citation_page_matches / citation_page_checks if citation_page_checks else None
+        ),
+        "citationPageChecks": citation_page_checks,
+        "filterCompliance": _mean(
+            1.0 if item.get("filterCompliant") and item.get("scopeCorrect", True) else 0.0
+            for item in results
+        ),
+        "retrievedFactCoverageAt5": _fact_coverage(results, "requiredFactsCoveredAt5"),
+        "retrievedFactCoverageAt10": _fact_coverage(results, "requiredFactsCoveredAt10"),
+        "usableEvidenceFactCoverage": _fact_coverage(results, "requiredFactsCovered"),
+        "requiredFactCoverage": _fact_coverage(results, "requiredFactsCoveredAt10"),
+        "forbiddenClaimsInPrompt": sum(len(item.get("forbiddenClaimsInPrompt", [])) for item in results),
+        "sectionCoverage": _section_coverage(results),

+    grouped = defaultdict(list)
+    for item in results:
+        grouped[str(item.get("domain") or "UNKNOWN")].append(item)

+def _grade_macro_f1(results: list[dict]) -> float:
+    scores = []
+    for label in sorted(VALID_GRADES):
+        true_positive = sum(
+            item.get("expectedGrade") == label and item.get("actualGrade") == label for item in results
+        )
+        false_positive = sum(
+            item.get("expectedGrade") != label and item.get("actualGrade") == label for item in results
+        )
+        false_negative = sum(
+            item.get("expectedGrade") == label and item.get("actualGrade") != label for item in results
+        )
+        precision = true_positive / max(1, true_positive + false_positive)
+        recall = true_positive / max(1, true_positive + false_negative)
+        scores.append(2 * precision * recall / max(1e-12, precision + recall))
+    return _mean(scores)

+def _distinct_documents(candidates, limit: int = RECALL_K) -> list:
+    seen = set()
+    rows = []
+    for item in candidates:
+        key = item.canonical_key or item.source_key
+        if key in seen:
+            continue
+        seen.add(key)
+        rows.append(item)
+    return rows[:limit]

+def _ndcg(keys: list[str], expected: set[str]) -> float:
+    if not expected:
+        return 1.0
+    gains = [1.0 if key in expected else 0.0 for key in keys[:TOP_K]]
+    dcg = sum(gain / math.log2(index + 2.0) for index, gain in enumerate(gains))
+    ideal_count = min(len(expected), TOP_K)
+    ideal = sum(1.0 / math.log2(index + 2.0) for index in range(ideal_count))
+    return dcg / ideal if ideal else 0.0
+

+def _covered_sections(candidates, expected_sections: list[str]) -> list[str]:
+    values = []
+    for section in expected_sections:
+        if any(section in str(item.section_title or "") for item in candidates):
+            values.append(section)
+    return values

+        for field in (
+            "canonical_key",
+            "title",
+            "source_url",
+            "section_title",
+            "site",
+            "content",
+        )
+    )

+def _fact_coverage(results: list[dict], field: str = "requiredFactsCovered") -> float:
+    expected = sum(len(item.get("requiredFacts", [])) for item in results)
+    covered = sum(len(item.get(field, [])) for item in results)

+def _mean(values) -> float:
+    rows = list(values)
+    return sum(rows) / len(rows) if rows else 1.0
+
+
+def _percentile(values: list[float], percentile: float) -> float:
+    if not values:
+        return 0.0
+    rows = sorted(values)
+    index = min(len(rows) - 1, max(0, math.ceil(len(rows) * percentile) - 1))
+    return rows[index]
+
+
+def corpus_chunk_metrics(db) -> dict:
+    lengths = [
+        len(row.content or "")
+        for row in db.query(KnowledgeChunk)
+        .join(KnowledgeDocument)
+        .filter(KnowledgeDocument.status == "ACTIVE")
+        .order_by(KnowledgeChunk.id.asc())
+        .all()
+    ]
+    if not lengths:
+        distribution = {"min": 0, "p50": 0, "p95": 0, "max": 0, "mean": 0.0}
+    else:
+        distribution = {
+            "min": min(lengths),
+            "p50": int(_percentile(lengths, 0.50)),
+            "p95": int(_percentile(lengths, 0.95)),
+            "max": max(lengths),
+            "mean": round(sum(lengths) / len(lengths), 3),
+        }
+    return {
+        "profile": _active_chunk_profile(db),
+        "unit": "python_characters",
+        "activeChunkCount": len(lengths),
+        "lengthCharacters": distribution,
+    }
+
+
+def _active_chunk_profile(db) -> str:
+    profiles = sorted({
+        str(value)
+        for (value,) in db.query(KnowledgeChunk.chunking_profile)
+        .join(KnowledgeDocument)
+        .filter(KnowledgeDocument.status == "ACTIVE")
+        .distinct()
+        .all()
+        if value
+    })
+    return profiles[0] if len(profiles) == 1 else "+".join(profiles) or "unknown"
+

+    path = Path(value)
